# Auto-detect GPU backend and extra features from glint_render_backend.h,
# which is generated by:  node third_party/glint/scripts/init_skia.mjs --backend <name>
# Manual overrides are still accepted via -D on the cmake command line.
set(_GLINT_BACKEND_HEADER "${CMAKE_CURRENT_LIST_DIR}/glint_render_backend.h")
set(_GLINT_AUTODETECT_GPU     OFF)
set(_GLINT_AUTODETECT_D3D12   OFF)
if(EXISTS "${_GLINT_BACKEND_HEADER}")
  file(STRINGS "${_GLINT_BACKEND_HEADER}" _glint_backend_lines)
  foreach(_line IN LISTS _glint_backend_lines)
    if(_line MATCHES "^#define GLINT_RENDER_GPU 1")
      set(_GLINT_AUTODETECT_GPU ON)
    endif()
    if(_line MATCHES "^#define GLINT_ENABLE_D3D12 1")
      set(_GLINT_AUTODETECT_D3D12 ON)
    endif()
  endforeach()
  message(STATUS "glint: glint_render_backend.h → GLINT_RENDER_GPU=${_GLINT_AUTODETECT_GPU}  GLINT_ENABLE_D3D12=${_GLINT_AUTODETECT_D3D12}")
else()
  message(STATUS "glint: glint_render_backend.h not found — defaulting to CPU rendering. "
                 "Run:  node third_party/glint/scripts/init_skia.mjs --backend <name>  to enable GPU.")
endif()

# macOS host target — links Skia libs from the computed _SKIA_LIB_DIR
# (GLINT_DEPS_DIR/mac/<arch>/<config>/, populated by init_skia.mjs --backend metal)
# plus Apple system frameworks.

  add_library(glint_host_mac_support STATIC
    "${CMAKE_CURRENT_LIST_DIR}/glint_standalone_stubs.cpp"
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_cursor_mac.mm"
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_window_mac.mm"
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_view_mac.mm")
  set_source_files_properties(
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_cursor_mac.mm"
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_window_mac.mm"
    "${CMAKE_CURRENT_LIST_DIR}/platform/mac/glint_view_mac.mm"
    PROPERTIES COMPILE_FLAGS "-x objective-c++")
  target_link_libraries(glint_host_mac_support PRIVATE glint::core)

  add_library(glint_host_mac INTERFACE)
  add_library(glint::host_mac ALIAS glint_host_mac)
  set_target_properties(glint_host_mac PROPERTIES EXPORT_NAME host_mac)
  target_link_libraries(glint_host_mac INTERFACE glint::core glint_host_mac_support)
  # The prebuilt libskia.a was compiled WITH SK_TRIVIAL_ABI=[[clang::trivial_abi]]
  # in the Debug config but WITHOUT it in the Release config.  That annotation
  # makes sk_sp<T> use register return on ARM64 (result in x0) instead of SRET
  # (hidden output pointer in x8).  Every TU that calls a Skia function returning
  # sk_sp<T> must see the same definition as the library it was linked against;
  # a mismatch means the caller reads x0 while Skia wrote to x8 (or vice-versa),
  # leaving the sk_sp holding a garbage pointer → crash at the first dereference.
  target_compile_definitions(glint_host_mac INTERFACE
    "$<$<CONFIG:Debug>:SK_TRIVIAL_ABI=[[clang::trivial_abi]]>")
  target_compile_definitions(glint_host_mac_support PRIVATE
    "$<$<CONFIG:Debug>:SK_TRIVIAL_ABI=[[clang::trivial_abi]]>")

  foreach(_skia_lib skia svg skshaper skparagraph skunicode_core skunicode_icu)
    target_link_libraries(glint_host_mac INTERFACE
      "${_SKIA_LIB_DIR}/lib${_skia_lib}.a")
  endforeach()

  target_link_libraries(glint_host_mac INTERFACE
    "-framework Metal"
    "-framework MetalKit"
    "-framework QuartzCore"
    "-framework CoreGraphics"
    "-framework CoreText"
    "-framework CoreFoundation"
    "-framework Foundation"
    "-framework AppKit"
    "-framework CoreVideo"
  )
endif()

# ── glint_configure_win32_app ─────────────────────────────────────────────────
# Embeds the glint Per-Monitor V2 DPI manifest into a Win32 executable target.
# Call this on every WIN32 executable that uses glint::host_win32:
#
#   glint_configure_win32_app(my_app)
#
# Without the manifest, Windows virtualises the process at 96 DPI so
# GetDpiForWindow always returns 96, mDpr stays 1.0, and the rendered UI
# is blurry at any display scale other than 100%.
if(WIN32)
  set(GLINT_WIN32_DPI_MANIFEST
    "${CMAKE_CURRENT_LIST_DIR}/platform/win32/glint_dpi.manifest"
    CACHE INTERNAL "Path to the glint Per-Monitor V2 DPI awareness manifest")

  function(glint_configure_win32_app target)
    target_sources(${target} PRIVATE "${GLINT_WIN32_DPI_MANIFEST}")
    set_source_files_properties("${GLINT_WIN32_DPI_MANIFEST}"
      TARGET_DIRECTORY ${target}
      PROPERTIES VS_TOOL_OVERRIDE "Manifest")
  endfunction()
else()
  function(glint_configure_win32_app target)
  endfunction()
endif()
